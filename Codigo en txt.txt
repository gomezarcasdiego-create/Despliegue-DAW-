        function cambiarFormulario() {
            const tipo = document.getElementById('tipoOperacion').value;
            const cEcuacion = document.getElementById('camposEcuacion');
            const cGeo = document.getElementById('camposGeometria');
            const gVal2 = document.getElementById('groupValor2');

            if (tipo === 'ecuacion_2grado') {
                cEcuacion.style.display = 'block';
                cGeo.style.display = 'none';
            } else if (tipo === 'pitagoras') {
                cEcuacion.style.display = 'none';
                cGeo.style.display = 'block';
                gVal2.style.display = 'block';
                document.getElementById('labelVal1').innerText = 'Cateto a:';
                document.getElementById('labelVal2').innerText = 'Cateto b:';
            } else if (tipo === 'area_circulo') {
                cEcuacion.style.display = 'none';
                cGeo.style.display = 'block';
                gVal2.style.display = 'none';
                document.getElementById('labelVal1').innerText = 'Radio (r):';
            } else if (tipo === 'area_triangulo') {
                cEcuacion.style.display = 'none';
                cGeo.style.display = 'block';
                gVal2.style.display = 'block';
                document.getElementById('labelVal1').innerText = 'Base (b):';
                document.getElementById('labelVal2').innerText = 'Altura (h):';
            }
        }

        document.getElementById('btnCalcular').addEventListener('click', async () => {
            const tipo = document.getElementById('tipoOperacion').value;
            const boxRes = document.getElementById('boxResultado');
            boxRes.style.display = 'none';

            let payload = { tipo: tipo };

            if (tipo === 'ecuacion_2grado') {
                payload.a = document.getElementById('inputA').value;
                payload.b = document.getElementById('inputB').value;
                payload.c = document.getElementById('inputC').value;
            } else if (tipo === 'pitagoras') {
                payload.a = document.getElementById('inputVal1').value;
                payload.b = document.getElementById('inputVal2').value;
            } else if (tipo === 'area_circulo') {
                payload.radio = document.getElementById('inputVal1').value;
            } else if (tipo === 'area_triangulo') {
                payload.base = document.getElementById('inputVal1').value;
                payload.altura = document.getElementById('inputVal2').value;
            }

            try {
                const response = await fetch('http://localhost:3000/api/calcular', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });

                const data = await response.json();

                boxRes.style.display = 'block';

                if (!response.ok) {
                    boxRes.className = 'resultado-box error';
                    document.getElementById('resTitulo').innerText = 'Error de cálculo';
                    document.getElementById('resTexto').innerText = data.error || 'Ocurrió un error inesperado.';
                    document.getElementById('resDetalle').innerText = '';
                    document.getElementById('resTime').innerText = '';
                } else {
                    boxRes.className = 'resultado-box success';
                    document.getElementById('resTitulo').innerText = 'Resultado Obtenido:';
                    document.getElementById('resTexto').innerText = data.resultado;
                    document.getElementById('resDetalle').innerText = data.detalle;
                    document.getElementById('resTime').innerText = 'Hora de procesamiento: ' + data.timestamp;
                }
            } catch (error) {
                boxRes.style.display = 'block';
                boxRes.className = 'resultado-box error';
                document.getElementById('resTitulo').innerText = 'Error de Conexión';
                document.getElementById('resTexto').innerText = 'No se pudo conectar con el servidor Node.js en http://localhost:3000';
                document.getElementById('resDetalle').innerText = 'Asegúrate de haber ejecutado "node app.js" en la terminal de VS Code.';
                document.getElementById('resTime').innerText = '';
            }
        });
    </script>

APP.JS




const express = require('express');
const app = express();
const PORT = 3000;

// Permite leer cuerpos JSON en las peticiones
app.use(express.json());

// Middleware CORS para permitir peticiones desde Apache (http://localhost)
app.use((req, res, next) => {
    res.header("Access-Control-Allow-Origin", "*");
    res.header("Access-Control-Allow-Headers", "Origin, X-Requested-With, Content-Type, Accept");
    res.header("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
    next();
});

// Ruta principal para verificar el estado de la API
app.get('/', (req, res) => {
    res.send('<h1>API Servidor Node.js / Express Activo</h1><p>Escuchando peticiones desde Apache.</p>');
});

// Ruta de cálculo matemático (API Backend)
app.post('/api/calcular', (req, res) => {
    const { tipo, a, b, c, radio, base, altura } = req.body;
    let resultado = null;
    let detalle = "";

    try {
        switch (tipo) {
            case 'ecuacion_2grado': {
                // ax² + bx + c = 0
                const numA = parseFloat(a);
                const numB = parseFloat(b);
                const numC = parseFloat(c);

                if (isNaN(numA) || isNaN(numB) || isNaN(numC)) {
                    return res.status(400).json({ error: "Parámetros no válidos para la ecuación." });
                }
                if (numA === 0) {
                    return res.status(400).json({ error: "El coeficiente 'a' no puede ser 0 en una ecuación de 2º grado." });
                }

                const discriminante = Math.pow(numB, 2) - (4 * numA * numC);

                if (discriminante < 0) {
                    resultado = "Sin soluciones reales";
                    detalle = `Discriminante (Δ = ${discriminante.toFixed(2)}) es negativo. Las raíces son complejas.`;
                } else if (discriminante === 0) {
                    const x = (-numB) / (2 * numA);
                    resultado = `x = ${x.toFixed(2)}`;
                    detalle = "Ecuación con solución única (raíz doble).";
                } else {
                    const x1 = ((-numB) + Math.sqrt(discriminante)) / (2 * numA);
                    const x2 = ((-numB) - Math.sqrt(discriminante)) / (2 * numA);
                    resultado = `x₁ = ${x1.toFixed(2)}, x₂ = ${x2.toFixed(2)}`;
                    detalle = `Discriminante Δ = ${discriminante.toFixed(2)}. La ecuación tiene 2 soluciones reales distintivas.`;
                }
                break;
            }

            case 'pitagoras': {
                const numA = parseFloat(a);
                const numB = parseFloat(b);
                if (isNaN(numA) || isNaN(numB) || numA <= 0 || numB <= 0) {
                    return res.status(400).json({ error: "Ingresa valores positivos válidos para los catetos." });
                }
                const hipotenusa = Math.sqrt(Math.pow(numA, 2) + Math.pow(numB, 2));
                resultado = `Hipotenusa (c) = ${hipotenusa.toFixed(2)}`;
                detalle = `Fórmula: c = √(${numA}² + ${numB}²) = √(${(numA**2 + numB**2).toFixed(2)})`;
                break;
            }

            case 'area_circulo': {
                const r = parseFloat(radio);
                if (isNaN(r) || r <= 0) {
                    return res.status(400).json({ error: "Proporciona un radio válido mayor que cero." });
                }
                const area = Math.PI * Math.pow(r, 2);
                const perimetro = 2 * Math.PI * r;
                resultado = `Área = ${area.toFixed(2)} u²`;
                detalle = `Perímetro = ${perimetro.toFixed(2)} u | Fórmula: A = π × ${r}²`;
                break;
            }

            case 'area_triangulo': {
                const b = parseFloat(base);
                const h = parseFloat(altura);
                if (isNaN(b) || isNaN(h) || b <= 0 || h <= 0) {
                    return res.status(400).json({ error: "Proporciona base y altura válidas." });
                }
                const area = (b * h) / 2;
                resultado = `Área = ${area.toFixed(2)} u²`;
                detalle = `Fórmula: A = (${b} × ${h}) / 2`;
                break;
            }

            default:
                return res.status(400).json({ error: "Tipo de operación matemática no reconocida." });
        }

        res.json({
            ok: true,
            tipo: tipo,
            resultado: resultado,
            detalle: detalle,
            timestamp: new Date().toLocaleTimeString()
        });

    } catch (err) {
        res.status(500).json({ error: "Error interno en el procesamiento matemático del servidor." });
    }
});

app.listen(PORT, () => {
    console.log(`====================================================`);
    console.log(`Servidor Node.js activo en: http://localhost:${PORT}`);
    console.log(`Listo para recibir datos desde Apache/XAMPP.`);
    console.log(`====================================================`);
});
